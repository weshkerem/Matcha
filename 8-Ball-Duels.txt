local function V(x,y) return {x=x,y=y} end
local function add(a,b) return V(a.x+b.x,a.y+b.y) end
local function sub(a,b) return V(a.x-b.x,a.y-b.y) end
local function mul(a,s) return V(a.x*s,a.y*s) end
local function dot(a,b) return a.x*b.x+a.y*b.y end
local function length(a) return math.sqrt(dot(a,a)) end
local function unit(a) local m=length(a);return m>1e-8 and mul(a,1/m) or V(0,0) end
local function reflect(d,n) return unit(sub(d,mul(n,2*dot(d,n)))) end
local function rayCircle(p,d,c,r)
 local q=sub(c,p);local along=dot(q,d)
 if along<=0 then return nil end
 local perp=dot(q,q)-along*along
 if perp>r*r then return nil end
 local t=along-math.sqrt(math.max(0,r*r-perp))
 if t>0.0001 then return t end
 return nil
end

local function pocket(a,b,side)
 local centre=mul(add(a,b),0.5)
 local tangent=unit(sub(b,a))
 local n=V(tangent.y,-tangent.x)
 if dot(n,centre)<0 then n=mul(n,-1) end
 local mouth=length(sub(b,a))
 local flare=side and math.rad(-14) or 0
 local radius=mouth/2
 local bowl=add(centre,mul(n,mouth*(side and 0.8 or 1.4)))
 local function facing(start)
  local outward=unit(sub(start,centre))
  local direction=add(mul(n,math.cos(flare)),mul(outward,math.sin(flare)))
  local q=sub(bowl,start)
  local projected=dot(q,direction)
  local inside=radius*radius-(dot(q,q)-projected*projected)
  if inside>0 then projected=projected-math.sqrt(inside) end
  local finish=add(start,mul(direction,math.max(projected,2/math.cos(flare))))
  local tang=unit(sub(finish,start))
  local normal=V(-tang.y,tang.x)
  if dot(normal,sub(centre,start))<0 then normal=mul(normal,-1) end
  rail(start,finish,normal)
  jaws[#jaws+1]=start
 end
 facing(a);facing(b)
 pockets[#pockets+1]={c=centre,n=n,mouth=mouth,depth=side and 0.3 or 2.6}
end

local function cast(p,d,r,balls,ignore,skip)
 local best={t=1000000,kind="end"}
 for _,b in ipairs(balls) do
  if b.id~=ignore and b.id~=skip then
   local t=rayCircle(p,d,b.p,r+b.r)
   if t and t<best.t then best={t=t,kind="ball",ball=b} end
  end
 end
 for _,s in ipairs(rails) do
  local approach=dot(d,s.n)
  if approach < -1e-7 then
   local distance=dot(sub(p,s.a),s.n)
   local t=(r-distance)/approach
   if distance>=-r and t>0.0001 and t<best.t then
    local at=add(p,mul(d,t));local delta=sub(s.b,s.a)
    local span=length(delta);local along=dot(sub(at,s.a),mul(delta,1/span))
    if along>=0 and along<=span then best={t=t,kind="rail",n=s.n} end
   end
  end
 end
 for _,j in ipairs(jaws) do
  local t=rayCircle(p,d,j,r)
  if t and t<best.t then best={t=t,kind="jaw",n=unit(sub(add(p,mul(d,t)),j))} end
 end
 for _,k in ipairs(pockets) do
  local approach=dot(d,k.n)
  if approach>1e-7 then
   local t=(k.depth-dot(sub(p,k.c),k.n))/approach
   if t>0.0001 and t<best.t then
    local at=sub(add(p,mul(d,t)),k.c)
    if math.abs(dot(at,V(-k.n.y,k.n.x)))<=k.mouth/2 then
     best={t=t,kind="pocket"}
    end
   end
  end
 end
 best.p=add(p,mul(d,best.t))
 return best
end

local function world(p) return Vector3.new(p.x,5001.125,-p.y) end
local segments,contact,finish={},nil,nil
local finishRadius=1.125
local function segment(a,b,color) segments[#segments+1]={a=a,b=b,color=color} end
local function trace(p,d,r,balls,id,skip,color,speed)
 for bounce=0,3 do
  local remaining=rollDistance(speed)
  if remaining<0.005 then return nil,d,speed end
  local hit=cast(p,d,r,balls,id,skip)
  if remaining<hit.t then
   local stop=add(p,mul(d,remaining))
   segment(p,stop,color)
   if color==yellow then finish=stop end
   return {kind="stop",p=stop},d,0
  end
  segment(p,hit.p,color)
  if color==yellow then finish=hit.p end
  speed=math.sqrt(math.max(0,speed*speed-2*physics.deceleration*hit.t))
  if hit.kind~="rail" and hit.kind~="jaw" then return hit,d,speed end
  if bounce==3 then return hit,d,speed end
  local normalSpeed=dot(d,hit.n)*speed
  local tangent=sub(mul(d,speed),mul(hit.n,normalSpeed))
  local velocity=sub(mul(tangent,1-physics.railFriction),mul(hit.n,normalSpeed*physics.railRestitution))
  speed=length(velocity);d=unit(velocity)
  p=add(hit.p,mul(d,0.002))
 end
end

local function predict()

    balls[#balls+1]=b
    if b.id==0 then white=b end

 local isBreak=#balls==16

  if b.id~=0 and (b.p.x<20 or b.p.x>32 or math.abs(b.p.y)>7) then isBreak=false end

 local function simulate(selectedPower)
  segments={};contact=nil;finish=nil
  local markerRadius=white.r
  local hit,incoming,impactSpeed=trace(white.p,d,white.r,balls,0,nil,yellow,shotSpeed(selectedPower)*multiplier)
  local targetNumber=nil
  if hit and hit.kind=="ball" then
   contact=hit.p
   local target=hit.ball
   targetNumber=target.id;markerRadius=target.r
   local normal=unit(sub(target.p,hit.p))
   local transfer=math.max(0,dot(incoming,normal))*(1+physics.ballRestitution)/2
   trace(target.p,normal,target.r,balls,target.id,0,yellow,impactSpeed*transfer)
  end
  return {segments=segments,contact=contact,finish=finish,radius=markerRadius,target=targetNumber}
 end

 segments=full.segments;contact=full.contact;finish=actual.finish;finishRadius=actual.radius
 H.Stats.target=full.target
 H.Stats.markerX=finish and finish.x
 H.Stats.markerY=finish and finish.y
 H.Stats.linePowerPercent=100

  local distance=0
  for _,s in ipairs(path.segments) do
   if s.color==yellow then distance=distance+length(sub(s.b,s.a)) end
  end
  return distance
 end

   lowX=low.finish and low.finish.x,lowY=low.finish and low.finish.y,

   lowTravel=yellowLength(low),highTravel=yellowLength(high)}
 end

end
